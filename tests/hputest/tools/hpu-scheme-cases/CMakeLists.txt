# 原04的Auto含专用twiddle及完整NTT链；main中的legacy demo仅做CPU预处理后的
# Galois KeySwitch，不能冒充同一覆盖。保持已验证legacy生产者及其STG语义。
if(DEFINED LEGACY_INLINE_ASM_ROOT)
  add_library(hpu_large_legacy_encode STATIC
    "${LEGACY_INLINE_ASM_ROOT}/encode/src/instruction.cpp"
    "${LEGACY_INLINE_ASM_ROOT}/encode/src/parser.cpp"
    "${LEGACY_INLINE_ASM_ROOT}/encode/src/encoder.cpp"
    "${LEGACY_INLINE_ASM_ROOT}/encode/src/assembler.cpp"
    "${LEGACY_INLINE_ASM_ROOT}/encode/src/executable.cpp")
  target_include_directories(hpu_large_legacy_encode PUBLIC "${LEGACY_INLINE_ASM_ROOT}/encode/include")
  target_compile_features(hpu_large_legacy_encode PUBLIC cxx_std_17)
  add_executable(hpu_render_legacy_program render_program.cpp)
  target_link_libraries(hpu_render_legacy_program PRIVATE hpu_large_legacy_encode)
  add_executable(hpu_legacy_program_model program_model.cpp)
  target_compile_definitions(hpu_legacy_program_model PRIVATE HPU_LEGACY_STG=1)
  target_link_libraries(hpu_legacy_program_model PRIVATE hpu_large_legacy_encode)
endif()
